# Interruption detection logic
class InterruptionHandler:
    def __init__(self):
        self.energy_threshold = 0.6  # Normalized audio energy
        self.intent_classifier = IntentClassifier()
        self.context_preserver = ContextPreserver()

    async def handle_barge_in(self, audio_frame, current_state):
        # 1. Detect energy spike (<50ms)
        if self.detect_energy_spike(audio_frame) > self.energy_threshold:

            # 2. Classify intent (question vs. statement vs. correction)
            intent = await self.intent_classifier.classify(audio_frame)

            # 3. Graceful fade-out (100ms ramp down)
            await self.fade_out_bot_audio(duration_ms=100)

            # 4. Preserve context for resumption
            self.context_preserver.save_state(current_state)

            # 5. Process interruption
            if intent == "correction":
                return await self.handle_correction(audio_frame)
            elif intent == "question":
                return await self.handle_question(audio_frame)
            elif intent == "new_order":
                return await self.handle_new_order(audio_frame)

-- Customers table
CREATE TABLE customers (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    phone_number VARCHAR(20) UNIQUE,
    email VARCHAR(255),
    name VARCHAR(100),
    preferences JSONB DEFAULT '{}',
    created_at TIMESTAMP DEFAULT NOW(),
    updated_at TIMESTAMP DEFAULT NOW()

-- Conversations table
CREATE TABLE conversations (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    customer_id UUID REFERENCES customers(id),
    session_id VARCHAR(100) UNIQUE NOT NULL,
    channel VARCHAR(20) CHECK (channel IN ('web', 'phone', 'kiosk', 'mobile')),
    started_at TIMESTAMP DEFAULT NOW(),
    ended_at TIMESTAMP,
    duration_seconds INTEGER,
    status VARCHAR(20) CHECK (status IN ('active', 'completed', 'escalated', 'abandoned')),
    sentiment_start FLOAT,
    sentiment_end FLOAT,
    metadata JSONB DEFAULT '{}'
);

-- Messages table (time-series)
CREATE TABLE messages (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    conversation_id UUID REFERENCES conversations(id),
    role VARCHAR(20) CHECK (role IN ('user', 'assistant', 'system')),
    content TEXT NOT NULL,
    audio_url VARCHAR(500),
    latency_ms INTEGER,
    tokens_input INTEGER,
    tokens_output INTEGER,
    timestamp TIMESTAMP DEFAULT NOW()
) PARTITION BY RANGE (timestamp);

-- Create monthly partitions
CREATE TABLE messages_y2025m01 PARTITION OF messages
    FOR VALUES FROM ('2025-01-01') TO ('2025-02-01');

-- Orders table
CREATE TABLE orders (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    conversation_id UUID REFERENCES conversations(id),
    customer_id UUID REFERENCES customers(id),
    items JSONB NOT NULL,
    total_amount DECIMAL(10,2) NOT NULL,
    tax_amount DECIMAL(10,2),
    status VARCHAR(20) CHECK (status IN ('pending', 'confirmed', 'preparing', 'ready', 'completed', 'cancelled')),
    estimated_ready_time TIMESTAMP,
    created_at TIMESTAMP DEFAULT NOW(),
    updated_at TIMESTAMP DEFAULT NOW()
);

-- Menu items table
CREATE TABLE menu_items (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    name VARCHAR(100) NOT NULL,
    category VARCHAR(50) CHECK (category IN ('wings', 'sides', 'drinks', 'dips', 'combos')),
    description TEXT,
    base_price DECIMAL(8,2),
    modifiers JSONB DEFAULT '[]',
    allergens TEXT[],
    nutrition JSONB,
    available BOOLEAN DEFAULT TRUE,
    embedding VECTOR(384)  -- For semantic search
);

-- Create indexes
CREATE INDEX idx_conversations_customer ON conversations(customer_id);
CREATE INDEX idx_conversations_session ON conversations(session_id);
CREATE INDEX idx_messages_conversation ON messages(conversation_id);
CREATE INDEX idx_orders_customer ON orders(customer_id);
CREATE INDEX idx_menu_embedding ON menu_items USING ivfflat (embedding vector_cosine_ops);

class UnderstandingEngine:
    def __init__(self):
        self.llm = GroqClient()
        self.ner_model = spaCy.load("en_core_web_sm")
        self.intent_classifier = IntentClassifier()

    async def understand(self, text: str, context: dict) -> Understanding:
        # Parallel processing
        intent_task = self.classify_intent(text)
        ner_task = self.extract_entities(text)
        sentiment_task = self.analyze_sentiment(text)

        intent, entities, sentiment = await asyncio.gather(
            intent_task, ner_task, sentiment_task
        )

        # LLM-based disambiguation
        if self.is_ambiguous(text, entities):
            clarification = await self.llm.generate(
                prompt=f"Clarify this order: {text}
Entities: {entities}"
            )
            entities = self.refine_entities(entities, clarification)

        return Understanding(
            intent=intent,
            entities=entities,
            sentiment=sentiment,
            urgency=self.calculate_urgency(intent, sentiment),
            ambiguity_score=self.calculate_ambiguity(text, entities),
            clarification_needed=len(entities) == 0
        )

class ToolRegistry:
    def __init__(self):
        self.tools = {
            "search_menu": MenuSearchTool(),
            "calculate_price": PriceCalculatorTool(),
            "create_order": OrderCreationTool(),
            "modify_order": OrderModificationTool(),
            "validate_order": OrderValidationTool(),
            "suggest_upsell": UpsellEngineTool(),
            "check_policy": PolicyCheckerTool(),
            "create_ticket": TicketCreationTool(),
            "escalate": HumanEscalationTool(),
        }

    async def execute(self, tool_name: str, params: dict) -> ToolResult:
        tool = self.tools.get(tool_name)
        if not tool:
            return ToolResult(success=False, error=f"Unknown tool: {tool_name}")

        # Circuit breaker pattern
        if tool.is_circuit_open():
            return ToolResult(success=False, error="Service temporarily unavailable")

        try:
            result = await tool.execute(**params)
            tool.record_success()
            return ToolResult(success=True, data=result)
        except Exception as e:
            tool.record_failure()
            return ToolResult(success=False, error=str(e))

class ResponseGenerator:
    def __init__(self):
        self.llm = GroqClient()
        self.templates = ResponseTemplates()

    async def generate(self, step: AgentStep, context: dict) -> str:
        # Use template for common responses (speed)
        if step.action.name in self.templates:
            template = self.templates.get(step.action.name)
            return template.fill(step.observation, context)

        # LLM generation for complex responses
        prompt = self.build_prompt(step, context)
        response = await self.llm.generate(
            prompt=prompt,
            max_tokens=60,  # ~20 words
            temperature=0.7
        )

        # Add prosody markers for TTS
        return self.add_prosody_markers(response)

    def add_prosody_markers(self, text: str) -> str:
        # Add SSML-like markers for TTS
        text = text.replace("!", '!<break time="200ms"/>')
        text = text.replace("?", '?<break time="300ms"/>')
        text = text.replace(",", ',<break time="150ms"/>')
        return f"<speak>{text}</speak>"

class TTSCache:
    def __init__(self):
        self.cache = LRUCache(maxsize=1000)
        self.s3 = S3Client()

    async def get_or_generate(self, text: str, voice_id: str) -> AudioSegment:
        cache_key = hashlib.md5(f"{text}:{voice_id}".encode()).hexdigest()

        # Check memory cache
        if cache_key in self.cache:
            return self.cache[cache_key]

        # Check S3 cache
        audio = await self.s3.get(f"tts-cache/{cache_key}.mp3")
        if audio:
            self.cache[cache_key] = audio
            return audio

        # Generate and cache
        audio = await self.tts.generate(text, voice_id)
        await self.s3.put(f"tts-cache/{cache_key}.mp3", audio)
        self.cache[cache_key] = audio
        return audio
